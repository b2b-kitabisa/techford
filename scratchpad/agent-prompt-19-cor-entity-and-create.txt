'use server'

import { revalidatePath } from 'next/cache'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'

export async function addCorEntity(formData: FormData) {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Operation') throw new Error('Role Operation tidak bisa menambah Vendor.')

  const entityName = ((formData.get('entity_name') as string) || '').trim()
  if (!entityName) throw new Error('Entity Name wajib diisi.')

  const admin = createAdminClient()
  const { error } = await admin.from('cor_entity').insert({
    entity_name: entityName,
    bank: (formData.get('bank') as string) || null,
    is_pkp: formData.get('is_pkp') === 'on',
    biaya_pencairan: Number(formData.get('biaya_pencairan') || 0),
    created_by: employee.name,
  })
  if (error) throw new Error(error.message)

  revalidatePath('/cor-entities')
}

'use server'

import { redirect } from 'next/navigation'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'

export async function createCorDocument(formData: FormData) {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Operation') throw new Error('Role Operation tidak bisa membuat COR.')

  const admin = createAdminClient()
  const projectId = (formData.get('project_id') as string) || null
  const vendorEntityId = (formData.get('vendor_entity_id') as string) || null

  const year = new Date().getFullYear() % 100
  const { data: seqValue, error: seqErr } = await admin.rpc('next_sequence', { seq_key: `DOC_${year}` })
  if (seqErr) throw new Error(seqErr.message)
  const docId = `DOC${year}-${String(seqValue).padStart(5, '0')}`

  const { data: doc, error: docErr } = await admin
    .from('document_pipeline')
    .insert({
      doc_id: docId,
      project_id: projectId,
      document_type: 'COR',
      status: 'Not Started',
      stage: 'New Request',
      requested_by: employee.name,
    })
    .select()
    .single()
  if (docErr || !doc) throw new Error(docErr?.message || 'Gagal membuat dokumen.')

  const isViaSalset = formData.get('is_via_salset') === 'on'
  const isSalsetOnly = formData.get('is_salset_only') === 'on'

  const { error: headerErr } = await admin.from('cor_header').insert({
    doc_id: doc.id,
    cor_method: 'GROSS_UP',
    is_via_salset: isViaSalset,
    is_salset_only: isSalsetOnly,
    vendor_entity_id: vendorEntityId,
    ngo_rate_pct: Number(formData.get('ngo_rate_pct') || 10),
    biaya_salset: Number(formData.get('biaya_salset') || 0),
    cor_tab: 'CLIENT',
    margin_enabled: true,
    margin_mode: 'COMPONENT',
    manual_project_name: projectId ? null : ((formData.get('manual_project_name') as string) || null),
    created_by: employee.name,
  })
  if (headerErr) throw new Error(headerErr.message)

  redirect('/documents')
}
