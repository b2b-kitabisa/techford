Add a PDF generator for COR and Quotation documents to this Next.js + Supabase app. Uses @react-pdf/renderer (pure Node/React PDF rendering, no headless browser needed -- works fine on Vercel serverless). Generated PDFs are uploaded to the private Supabase Storage "documents" bucket (already set up) using the existing src/lib/storage.ts helpers, at the path `<Client_ID>-<Brand_Name>/<Project_ID>/<file_name>.pdf` -- same folder-per-client/folder-per-project convention the old Google Drive version used, just as a storage path prefix instead of real folders. Downloading always goes through a route that mints a fresh short-lived signed URL (the bucket is private, so there's no permanent public link).

First install the PDF library: run `npm install @react-pdf/renderer` in the terminal.

// Matches the old Google Drive folder-name sanitizer: strip characters that
// are illegal in file/folder names, collapse whitespace, trim.
export function sanitizeName(name: string): string {
  return name
    .replace(/[/\\:*?"<>|]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim()
}

export function rupiah(n: number | null | undefined): string {
  if (n === null || n === undefined || Number.isNaN(n)) return '-'
  return 'Rp ' + Math.round(n).toLocaleString('id-ID')
}

export function percent(n: number | null | undefined): string {
  if (n === null || n === undefined || Number.isNaN(n)) return '-'
  return (Math.round(n * 10000) / 100).toString() + '%'
}

export async function generateCorPdf(formData: FormData) {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Operation') throw new Error('Role Operation tidak bisa generate PDF.')

  const docId = formData.get('doc_id') as string
  const admin = createAdminClient()

  const { data: doc } = await admin.from('document_pipeline').select('*').eq('id', docId).single()
  if (!doc) throw new Error('Dokumen tidak ditemukan.')

  const { header, vendor, fundRows, costs, marginComponents, result } = await loadCorComputation(admin, docId)
  if (!header) throw new Error('COR header tidak ditemukan.')

  let clientName = '-'
  let projectName = '-'
  let folderPrefix = '_unlinked'

  if (doc.project_id) {
    const { data: project } = await admin.from('project').select('project_id, project_name, client_id').eq('id', doc.project_id).single()
    if (project) {
      projectName = project.project_name
      const { data: client } = await admin.from('client').select('client_id, brand_name').eq('id', project.client_id).single()
      const clientFolder = client ? sanitizeName(`${client.client_id}-${client.brand_name}`) : 'unknown-client'
      clientName = client?.brand_name || '-'
      folderPrefix = `${clientFolder}/${sanitizeName(project.project_id)}`
    }
  }

  const pdfBuffer = await renderToBuffer(
    CorPdfDocument({
      docId: doc.doc_id,
      clientName,
      projectName,
      vendorName: (vendor as { entity_name?: string } | null)?.entity_name ?? '-',
      corMethod: header.cor_method,
      isViaSalset: header.is_via_salset,
      fundRows,
      costs,
      marginComponents,
      result,
    })
  )

  const fileName = `COR - ${doc.doc_id}.pdf`
  const path = `${folderPrefix}/${fileName}`
  await uploadDocumentFile(admin, path, pdfBuffer, 'application/pdf')

  await admin.from('cor_header').update({ pdf_storage_path: path, pdf_generated_at: new Date().toISOString() }).eq('doc_id', docId)

  await logAudit(admin, { actorUserId: employee.auth_user_id, actorEmail: employee.email, action: 'GENERATE_COR_PDF', entityType: 'cor_header', entityId: docId })

  revalidatePath(`/documents/cor/${docId}`)
}

import { NextResponse } from 'next/server'

export async function GET(_req: Request, { params }: { params: Promise<{ docId: string }> }) {
  const { docId } = await params
  const employee = await getCurrentEmployee()
  if (!employee) return NextResponse.redirect(new URL('/login', _req.url))

  const admin = createAdminClient()
  const { data: header } = await admin.from('cor_header').select('pdf_storage_path').eq('doc_id', docId).single()
  if (!header?.pdf_storage_path) {
    return new NextResponse('PDF belum pernah di-generate untuk dokumen ini.', { status: 404 })
  }

  const url = await getDocumentSignedUrl(admin, header.pdf_storage_path, 300)
  return NextResponse.redirect(url)
}

      items: items ?? [],
      total,
    })
  )

  await uploadDocumentFile(admin, path, pdfBuffer, 'application/pdf')

  await admin.from('quotation_header').update({ pdf_storage_path: path, pdf_generated_at: new Date().toISOString() }).eq('doc_id', docId)

  await logAudit(admin, { actorUserId: employee.auth_user_id, actorEmail: employee.email, action: 'GENERATE_QUOTATION_PDF', entityType: 'quotation_header', entityId: docId })

  revalidatePath(`/documents/quotation/${docId}`)
}

import { NextResponse } from 'next/server'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'
import { getDocumentSignedUrl } from '@/lib/storage'

export async function GET(_req: Request, { params }: { params: Promise<{ docId: string }> }) {
  const { docId } = await params
  const employee = await getCurrentEmployee()
  if (!employee) return NextResponse.redirect(new URL('/login', _req.url))

  const admin = createAdminClient()
  const { data: header } = await admin.from('quotation_header').select('pdf_storage_path').eq('doc_id', docId).single()
  if (!header?.pdf_storage_path) {
    return new NextResponse('PDF belum pernah di-generate untuk dokumen ini.', { status: 404 })
  }

  const url = await getDocumentSignedUrl(admin, header.pdf_storage_path, 300)
  return NextResponse.redirect(url)
}
