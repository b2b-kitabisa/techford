export async function addCost(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string
  const harga = Number(formData.get('harga') || 0)
  if (harga <= 0) throw new Error('Harga harus lebih dari 0.')

  const { error } = await admin.from('cor_cost').insert({
    doc_id: docId,
    cor_tab: 'CLIENT',
    cost_group: formData.get('cost_group') as string,
    row_role: 'PRICE',
    keterangan: (formData.get('keterangan') as string) || null,
    kategori: (formData.get('kategori') as string) || null,
    tipe: (formData.get('tipe') as string) || null,
    harga,
    qty: Number(formData.get('qty') || 1),
    periode: Number(formData.get('periode') || 1),
  })
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/cor/${docId}`)
}

export async function removeCost(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const id = formData.get('id') as string
  const docId = formData.get('doc_id') as string

  const { error } = await admin.from('cor_cost').delete().eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/cor/${docId}`)
}

export async function setMarginComponent(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string
  const component = formData.get('component') as string
  const percentage = Number(formData.get('percentage') || 0)
  const subCategory = (formData.get('sub_category') as string) || component

  await admin.from('cor_margin').delete().eq('doc_id', docId).eq('cor_tab', 'CLIENT').eq('component', component)

  const { error } = await admin.from('cor_margin').insert({
    doc_id: docId,
    cor_tab: 'CLIENT',
    component,
    sub_category: subCategory,
    percentage,
  })
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/cor/${docId}`)
}
