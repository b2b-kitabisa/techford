import { createAdminClient } from '@/lib/supabase/admin'

export type DuplicateRef = {
  id: string
  inbound_id: string
  entity_name: string
  status: string
  matched_by: 'email' | 'phone'
}

// Builds a map of leadId -> the other leads that share its (normalized)
// email or phone number. Computed over the WHOLE table, not just the
// currently filtered/displayed page, since a duplicate could easily fall
// outside the current filter.
export async function getDuplicateMap(admin: ReturnType<typeof createAdminClient>): Promise<Map<string, DuplicateRef[]>> {
  const { data: rows } = await admin.from('lead').select('id, inbound_id, entity_name, status, email, phone')
  const all = rows ?? []

  const byEmail = new Map<string, typeof all>()
  const byPhone = new Map<string, typeof all>()
  for (const l of all) {
    if (l.email) {
      const key = l.email.trim().toLowerCase()
      byEmail.set(key, [...(byEmail.get(key) || []), l])
    }
    if (l.phone) {
      const key = l.phone.replace(/\D/g, '')
      if (key) byPhone.set(key, [...(byPhone.get(key) || []), l])
    }
  }

  const result = new Map<string, DuplicateRef[]>()
  for (const l of all) {
    const refs = new Map<string, DuplicateRef>()
    if (l.email) {
      const key = l.email.trim().toLowerCase()
      for (const other of byEmail.get(key) || []) {
        if (other.id !== l.id) refs.set(other.id, { id: other.id, inbound_id: other.inbound_id, entity_name: other.entity_name, status: other.status, matched_by: 'email' })
      }
    }
    if (l.phone) {
      const key = l.phone.replace(/\D/g, '')
      if (key) {
        for (const other of byPhone.get(key) || []) {
          if (other.id !== l.id && !refs.has(other.id)) {
            refs.set(other.id, { id: other.id, inbound_id: other.inbound_id, entity_name: other.entity_name, status: other.status, matched_by: 'phone' })
          }
        }
      }
    }
    if (refs.size > 0) result.set(l.id, Array.from(refs.values()))
  }
  return result
}

function initials(name: string | null): string {
  if (!name) return '?'
  const parts = name.trim().split(/\s+/).filter(Boolean)
  return parts.slice(0, 2).map((p) => p[0]?.toUpperCase() || '').join('') || '?'
}

function CopyButton({ text }: { text: string }) {
  const [copied, setCopied] = useState(false)
  function handleCopy(e: React.MouseEvent) {
    e.stopPropagation()
    if (!navigator.clipboard) return
    navigator.clipboard
      .writeText(text)
      .then(() => {
        setCopied(true)
        setTimeout(() => setCopied(false), 1500)
      })
      .catch(() => {})
  }
  return (
    <button onClick={handleCopy} title={copied ? 'Tersalin!' : 'Copy'} className="inline-flex shrink-0 items-center text-gray-400 hover:text-gray-700">
      <CopyIcon />
    </button>
  )
}

export function LeadsTable({ leads, canWrite }: { leads: Lead[]; canWrite: boolean }) {
  const [selected, setSelected] = useState<Set<string>>(new Set())
  const [activeIndex, setActiveIndex] = useState<number | null>(null)
  const [isPending, startTransition] = useTransition()
  const router = useRouter()

  function toggleOne(id: string) {
    setSelected((prev) => {
      const next = new Set(prev)
      if (next.has(id)) next.delete(id)
      else next.add(id)
      return next
    })
  }

  function toggleAll() {
    setSelected((prev) => (prev.size === leads.length ? new Set() : new Set(leads.map((l) => l.id))))
  }

  function runBulk(status: string) {
    startTransition(async () => {
      await bulkUpdateLeadStatus(Array.from(selected), status)
      setSelected(new Set())
      router.refresh()
    })
  }

  return (
    <div>
      {canWrite && selected.size > 0 && (
        <div className="mb-3 flex items-center justify-between rounded-lg bg-gray-900 px-4 py-2 text-sm text-white">
          <span>{selected.size} lead dipilih</span>

              Tandai Other
            </button>
            <button disabled={isPending} onClick={() => runBulk('Spam')} className="rounded-md bg-white/10 px-3 py-1 hover:bg-white/20 disabled:opacity-50">

                  className="rounded-md bg-indigo-600 px-3 py-1.5 text-xs font-medium text-white hover:bg-indigo-500 disabled:opacity-50"
                >
                  &raquo; Move to Client
                </button>
              </form>
            </div>
          )}
          {lead.status === 'Moved' && <p className="mt-4 text-xs italic text-gray-400">Terkunci (sudah jadi Client).</p>}
        </div>

        <div className="flex-1 overflow-y-auto p-5">
          <div className="mb-5 flex items-center gap-3">
            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-blue-500 text-sm font-semibold text-white">
              {initials(lead.pic_name)}
            </div>
            <div className="min-w-0 flex-1">
              <p className="truncate text-sm font-semibold text-gray-900">{lead.pic_name || '-'}</p>
              {lead.phone && (
                <div className="flex items-center gap-1">
                  <a href={`tel:${lead.phone}`} className="truncate text-xs text-blue-600 hover:underline">
                    {formatPhone(lead.phone)}
                  </a>
                  <CopyButton text={formatPhone(lead.phone)} />
                </div>
              )}
              {lead.email && (
                <div className="flex items-center gap-1">
                  <a href={`mailto:${lead.email}`} className="truncate text-xs text-blue-600 hover:underline">
                    {lead.email}
                  </a>
                  <CopyButton text={lead.email} />
                </div>
              )}
            </div>
          </div>

          {lead.duplicates.length > 0 && (
            <div className="mb-4">
              <DuplicateSection duplicates={lead.duplicates} onDone={() => router.refresh()} />
            </div>
          )}

          <div className="border-t pt-4">
            <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-gray-400">Interest</p>
            <div className="mb-3">
              <p className="text-xs font-semibold text-gray-500">Kebutuhan</p>
              <p className="mt-0.5 text-sm text-gray-800">{lead.detail_interest || '-'}</p>
            </div>
            <div>
              <p className="text-xs font-semibold text-gray-500">Prioritas</p>
              <p className="mt-0.5 text-sm text-gray-800">{lead.priority_notes || '-'}</p>
            </div>
          </div>

          {trafficTags.length > 0 && (
            <div className="mt-4 border-t pt-4">
              <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-gray-400">Sumber Trafik</p>
              <div className="flex flex-wrap gap-2">
                {trafficTags.map((t) => (
                  <span key={t.label} className="rounded-full border border-gray-200 bg-gray-50 px-3 py-1 text-xs text-gray-600">
                    {t.label} <span className="font-semibold text-gray-800">{t.value}</span>
                  </span>
                ))}
              </div>
            </div>
          )}

          <div className="mt-4 border-t pt-4">
            <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-gray-400">Other Notes</p>
            <p className="text-sm text-gray-600">{lead.other_notes || 'Belum ada catatan.'}</p>
          </div>
        </div>
      </div>
    </div>
  )
}

function DuplicateSection({ duplicates, onDone }: { duplicates: DuplicateRef[]; onDone: () => void }) {
  const [selected, setSelected] = useState<Set<string>>(new Set(duplicates.map((d) => d.id)))
  const [bulkStatus, setBulkStatus] = useState('Spam')
  const [isPending, startTransition] = useTransition()

  function toggle(id: string) {
    setSelected((prev) => {
      const next = new Set(prev)
      if (next.has(id)) next.delete(id)
      else next.add(id)
      return next
    })
  }

  function applyBulk() {
    startTransition(async () => {
      await bulkUpdateLeadStatus(Array.from(selected), bulkStatus)
      onDone()
    })
  }

  return (
    <details className="rounded-md border border-orange-200 bg-orange-50" open>
      <summary className="cursor-pointer px-3 py-2 text-xs font-semibold uppercase tracking-wide text-orange-700">
        Potensi Duplikat ({duplicates.length})
      </summary>
      <div className="space-y-2 border-t border-orange-200 px-3 py-2">
        {duplicates.map((d) => (
          <label key={d.id} className="flex items-center gap-2 text-xs text-gray-700">
            <input type="checkbox" checked={selected.has(d.id)} onChange={() => toggle(d.id)} />
            <span className="flex-1 truncate">
              {d.entity_name} <span className="text-gray-400">({d.inbound_id})</span>
            </span>
            <span className={`rounded-full px-1.5 py-0.5 text-[10px] font-medium ${STATUS_STYLES[d.status] ?? 'bg-gray-100 text-gray-700'}`}>{d.status}</span>
            <span className="text-[10px] text-gray-400">via {d.matched_by === 'email' ? 'Email' : 'Phone'}</span>
          </label>
        ))}
        <div className="flex items-center gap-2 pt-2">
          <select value={bulkStatus} onChange={(e) => setBulkStatus(e.target.value)} className="rounded-md border-gray-300 text-xs">
            {STATUS_OPTIONS.map((s) => (
              <option key={s} value={s}>
                {s}
              </option>
            ))}
          </select>
          <button
            onClick={applyBulk}
            disabled={isPending || selected.size === 0}
            className="rounded-md bg-orange-600 px-3 py-1.5 text-xs font-medium text-white hover:bg-orange-500 disabled:opacity-50"
          >
            Update {selected.size} Data Terpilih
          </button>
        </div>
      </div>
    </details>
  )
}
