Add a Cost Monitoring module to this Next.js + Supabase app. IMPORTANT: unlike other modules where Operation is view-only, in THIS module Consultant is the view-only role (Master Admin, Operation, and Head of B2B can all write here). It tracks realized disbursements (cor_disbursement) against the frozen budget snapshot (cor_budget_item) that was created when a COR got approved.

'use server'

import { revalidatePath } from 'next/cache'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'

async function requireWriteAccess() {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Consultant') throw new Error('Role Consultant cuma bisa lihat Cost Monitoring, tidak bisa mengubah.')
  return employee
}

export async function addDisbursement(formData: FormData) {
  const employee = await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string
  const budgetItemId = formData.get('budget_item_id') as string
  const amount = Number(formData.get('amount') || 0)

  if (amount <= 0) throw new Error('Amount harus lebih dari 0.')

  const { data: header } = await admin.from('cor_header').select('cost_monitoring_closed').eq('doc_id', docId).single()
  if (header?.cost_monitoring_closed) throw new Error('Cost Monitoring untuk dokumen ini sudah ditutup, tidak bisa menambah realisasi lagi.')

  const { error } = await admin.from('cor_disbursement').insert({
    doc_id: docId,
    budget_item_id: budgetItemId,
    amount,
    disbursement_date: (formData.get('disbursement_date') as string) || new Date().toISOString().slice(0, 10),
    note: (formData.get('note') as string) || null,
    created_by: employee.name,
  })
  if (error) throw new Error(error.message)

  revalidatePath(`/cost-monitoring/${docId}`)
  revalidatePath('/cost-monitoring')
}

export async function closeCostMonitoring(formData: FormData) {
  const employee = await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string

  const { error } = await admin.from('cor_header').update({
    cost_monitoring_closed: true,
    cost_monitoring_closed_by: employee.id,
    cost_monitoring_closed_at: new Date().toISOString(),
  }).eq('doc_id', docId)
  if (error) throw new Error(error.message)

  revalidatePath(`/cost-monitoring/${docId}`)
  revalidatePath('/cost-monitoring')
}

import { redirect } from 'next/navigation'
import Link from 'next/link'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'
import { AppShell } from '@/components/app-shell'

function rupiah(n: number) {
  return 'Rp ' + Math.round(n).toLocaleString('id-ID')
}

export default async function CostMonitoringPage() {
  const employee = await getCurrentEmployee()
  if (!employee) redirect('/login')

  const admin = createAdminClient()
  const { data: budgetItems } = await admin.from('cor_budget_item').select('doc_id, budgeted_amount')
  const docIds = Array.from(new Set((budgetItems ?? []).map((b) => b.doc_id)))

  const { data: headers } = docIds.length
    ? await admin.from('cor_header').select('*').in('doc_id', docIds)
    : { data: [] }
  const { data: docs } = docIds.length
    ? await admin.from('document_pipeline').select('*').in('id', docIds)
    : { data: [] }
  const { data: disbursements } = docIds.length
    ? await admin.from('cor_disbursement').select('doc_id, amount').in('doc_id', docIds)
    : { data: [] }

  const rows = docIds.map((docId) => {
    const doc = (docs ?? []).find((d) => d.id === docId)
    const header = (headers ?? []).find((h) => h.doc_id === docId)
    const totalBudget = (budgetItems ?? []).filter((b) => b.doc_id === docId).reduce((sum, b) => sum + Number(b.budgeted_amount), 0)
    const totalRealized = (disbursements ?? []).filter((d) => d.doc_id === docId).reduce((sum, d) => sum + Number(d.amount), 0)
    let label = 'Belum Ada Realisasi'
    if (header?.cost_monitoring_closed) label = 'Selesai'
    else if (totalRealized > totalBudget) label = 'Melebihi Anggaran'
    else if (totalRealized > 0) label = 'Dalam Proses'
    return { docId, docCode: doc?.doc_id, totalBudget, totalRealized, label, closed: header?.cost_monitoring_closed }
  })

  const LABEL_STYLES: Record<string, string> = {
    'Belum Ada Realisasi': 'bg-gray-100 text-gray-700',
    'Dalam Proses': 'bg-amber-100 text-amber-700',
    'Melebihi Anggaran': 'bg-red-100 text-red-700',
    'Selesai': 'bg-green-100 text-green-700',
  }

  return (
    <AppShell employeeName={employee.name} employeeRole={employee.role}>
      <h1 className="mb-6 text-2xl font-bold text-gray-900">Cost Monitoring</h1>
      <div className="overflow-hidden rounded-lg border bg-white shadow-sm">
        <table className="w-full text-sm">
          <thead className="bg-gray-50 text-left text-xs font-semibold uppercase tracking-wide text-gray-500">
            <tr><th className="px-4 py-3">Doc ID</th><th className="px-4 py-3">Budget</th><th className="px-4 py-3">Realisasi</th><th className="px-4 py-3">Status</th><th className="px-4 py-3"></th></tr>
          </thead>
          <tbody className="divide-y">
            {rows.map((r) => (
              <tr key={r.docId}>
                <td className="px-4 py-3 font-mono text-gray-700">{r.docCode}</td>
                <td className="px-4 py-3 text-gray-700">{rupiah(r.totalBudget)}</td>
                <td className="px-4 py-3 text-gray-700">{rupiah(r.totalRealized)}</td>
                <td className="px-4 py-3"><span className={`rounded-full px-2 py-0.5 text-xs font-medium ${LABEL_STYLES[r.label]}`}>{r.label}</span></td>
                <td className="px-4 py-3"><Link href={`/cost-monitoring/${r.docId}`} className="text-xs font-medium text-blue-600 hover:text-blue-700">Detail</Link></td>
              </tr>
            ))}
            {rows.length === 0 && (
              <tr><td colSpan={5} className="px-4 py-8 text-center text-gray-400">Belum ada COR yang di-approve.</td></tr>
            )}
          </tbody>
        </table>
      </div>
    </AppShell>
  )
}

import { redirect, notFound } from 'next/navigation'
import Link from 'next/link'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'
import { AppShell } from '@/components/app-shell'
import { addDisbursement, closeCostMonitoring } from '../actions'

function rupiah(n: number) {
  return 'Rp ' + Math.round(n).toLocaleString('id-ID')
}

export default async function CostMonitoringDetailPage({ params }: { params: Promise<{ docId: string }> }) {
  const { docId } = await params
  const employee = await getCurrentEmployee()
  if (!employee) redirect('/login')

  const admin = createAdminClient()
  const { data: doc } = await admin.from('document_pipeline').select('*').eq('id', docId).single()
  if (!doc) notFound()
  const { data: header } = await admin.from('cor_header').select('*').eq('doc_id', docId).single()
  const { data: budgetItems } = await admin.from('cor_budget_item').select('*').eq('doc_id', docId).order('sort_order')
  const { data: disbursements } = await admin.from('cor_disbursement').select('*').eq('doc_id', docId).order('created_at', { ascending: false })

  const canWrite = employee.role !== 'Consultant'
  const totalBudget = (budgetItems ?? []).reduce((sum, b) => sum + Number(b.budgeted_amount), 0)
  const totalRealized = (disbursements ?? []).reduce((sum, d) => sum + Number(d.amount), 0)
  const realizedByItem = (id: string) => (disbursements ?? []).filter((d) => d.budget_item_id === id).reduce((sum, d) => sum + Number(d.amount), 0)

  return (
    <AppShell employeeName={employee.name} employeeRole={employee.role}>
      <p className="mb-2"><Link href="/cost-monitoring" className="text-sm font-medium text-gray-600 hover:text-gray-900">&larr; Cost Monitoring</Link></p>
      <div className="mb-6 flex items-center justify-between">
        <h1 className="text-2xl font-bold text-gray-900">COR: {doc.doc_id}</h1>
        {canWrite && !header?.cost_monitoring_closed && (
          <form action={closeCostMonitoring}>
            <input type="hidden" name="doc_id" value={docId} />
            <button type="submit" className="rounded-md border border-red-300 bg-red-50 px-4 py-2 text-sm font-medium text-red-700 hover:bg-red-100">
              Tutup Cost Monitoring (satu arah)
            </button>
          </form>
        )}
        {header?.cost_monitoring_closed && (
          <span className="rounded-full bg-green-100 px-3 py-1 text-xs font-medium text-green-700">Sudah Ditutup</span>
        )}
      </div>

      <div className="mb-6 grid gap-4 sm:grid-cols-2">
        <div className="rounded-lg border bg-white p-4 shadow-sm">
          <div className="text-xs font-medium uppercase tracking-wide text-gray-500">Total Budget</div>
          <div className="mt-1 text-xl font-bold text-gray-900">{rupiah(totalBudget)}</div>
        </div>
        <div className="rounded-lg border bg-white p-4 shadow-sm">
          <div className="text-xs font-medium uppercase tracking-wide text-gray-500">Total Realisasi</div>
          <div className={`mt-1 text-xl font-bold ${totalRealized > totalBudget ? 'text-red-600' : 'text-gray-900'}`}>{rupiah(totalRealized)}</div>
        </div>
      </div>

      <h2 className="mb-3 text-lg font-semibold text-gray-900">Budget Items</h2>
      <div className="mb-8 overflow-hidden rounded-lg border bg-white shadow-sm">
        <table className="w-full text-sm">
          <thead className="bg-gray-50 text-left text-xs font-semibold uppercase tracking-wide text-gray-500">
            <tr><th className="px-4 py-3">Keterangan</th><th className="px-4 py-3">Kelompok</th><th className="px-4 py-3">Budgeted</th><th className="px-4 py-3">Realized</th><th className="px-4 py-3"></th></tr>
          </thead>
          <tbody className="divide-y">
            {(budgetItems ?? []).map((b) => (
              <tr key={b.id}>
                <td className="px-4 py-3 text-gray-900">{b.keterangan || '-'}</td>
                <td className="px-4 py-3 text-gray-600">{b.cost_group}</td>
                <td className="px-4 py-3 text-gray-700">{rupiah(b.budgeted_amount)}</td>
                <td className="px-4 py-3 text-gray-700">{rupiah(realizedByItem(b.id))}</td>
                <td className="px-4 py-3">
                  {canWrite && !header?.cost_monitoring_closed && (
                    <details>
                      <summary className="cursor-pointer text-xs font-medium text-blue-600 hover:text-blue-700">+ Realisasi</summary>
                      <form action={addDisbursement} className="mt-2 flex flex-col gap-2 rounded-md border bg-gray-50 p-2">
                        <input type="hidden" name="doc_id" value={docId} />
                        <input type="hidden" name="budget_item_id" value={b.id} />
                        <input name="amount" type="number" placeholder="Amount" required className="rounded-md border-gray-300 text-xs" />
                        <input name="disbursement_date" type="date" className="rounded-md border-gray-300 text-xs" />
                        <input name="note" placeholder="Catatan" className="rounded-md border-gray-300 text-xs" />
                        <button type="submit" className="rounded-md bg-gray-900 px-3 py-1 text-xs font-medium text-white">Simpan</button>
                      </form>
                    </details>
                  )}
                </td>
              </tr>
            ))}
            {(budgetItems ?? []).length === 0 && (
              <tr><td colSpan={5} className="px-4 py-8 text-center text-gray-400">Belum ada budget item.</td></tr>
            )}
          </tbody>
        </table>
      </div>

      <h2 className="mb-3 text-lg font-semibold text-gray-900">Riwayat Realisasi</h2>
      <div className="overflow-hidden rounded-lg border bg-white shadow-sm">
        <table className="w-full text-sm">
          <thead className="bg-gray-50 text-left text-xs font-semibold uppercase tracking-wide text-gray-500">
            <tr><th className="px-4 py-3">Tanggal</th><th className="px-4 py-3">Amount</th><th className="px-4 py-3">Catatan</th><th className="px-4 py-3">Oleh</th></tr>
          </thead>
          <tbody className="divide-y">
            {(disbursements ?? []).map((d) => (
              <tr key={d.id}>
                <td className="px-4 py-3 text-gray-700">{d.disbursement_date}</td>
                <td className="px-4 py-3 text-gray-700">{rupiah(d.amount)}</td>
                <td className="px-4 py-3 text-gray-600">{d.note || '-'}</td>
                <td className="px-4 py-3 text-gray-600">{d.created_by}</td>

            )}
          </tbody>
        </table>
      </div>
    </AppShell>
  )
}
