'use server'

import { revalidatePath } from 'next/cache'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'

async function requireWriteAccess() {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Operation') throw new Error('Role Operation tidak bisa mengubah COR.')
  return employee
}

export async function addFund(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string
  const fundType = formData.get('fund_type') as string
  const gdv = Number(formData.get('gdv') || 0)
  if (gdv <= 0) throw new Error('Nominal harus lebih dari 0.')

  const { error } = await admin.from('cor_fund').insert({
    doc_id: docId,
    fund_type: fundType,
    link_campaign: (formData.get('link_campaign') as string) || null,
    campaign_fund_kind: fundType === 'CAMPAIGN' ? ((formData.get('campaign_fund_kind') as string) || null) : null,
    gdv,
    is_zakat: formData.get('is_zakat') === 'on',
    tech_fee_manual: formData.get('tech_fee_manual') === 'on',
    manual_tech_fee: formData.get('tech_fee_manual') === 'on' ? Number(formData.get('manual_tech_fee') || 0) : null,
  })
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/cor/${docId}`)
}

export async function removeFund(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const id = formData.get('id') as string
  const docId = formData.get('doc_id') as string

  const { error } = await admin.from('cor_fund').delete().eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/cor/${docId}`)
}
