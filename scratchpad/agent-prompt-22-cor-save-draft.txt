import { fundCalc, calcItemRow, computeGD, computeGU } from './cor-calc'

const MARGIN_COMPONENTS_LIST = ['CONS', 'CRE', 'PROG', 'IMP']

export async function loadCorComputation(admin: any, docId: string) {
  const { data: header } = await admin.from('cor_header').select('*').eq('doc_id', docId).single()
  const { data: funds } = await admin.from('cor_fund').select('*').eq('doc_id', docId).order('sort_order')
  const { data: costs } = await admin.from('cor_cost').select('*').eq('doc_id', docId).order('sort_order')
  const { data: margins } = await admin.from('cor_margin').select('*').eq('doc_id', docId).eq('cor_tab', 'CLIENT')
  const { data: vendor } = header?.vendor_entity_id
    ? await admin.from('cor_entity').select('*').eq('id', header.vendor_entity_id).single()
    : { data: null }

  const biayaPencairan = vendor?.biaya_pencairan ?? 0
  const pkp = vendor?.is_pkp ?? false

  const fundRows = (funds ?? []).map((f: any) => ({
    ...f,
    calc: fundCalc(
      { fundType: f.fund_type, nominal: Number(f.gdv), isZakat: f.is_zakat, techFeeManual: f.tech_fee_manual, manualTechFee: Number(f.manual_tech_fee || 0) },
      biayaPencairan
    ),
  }))

  const salItems = (costs ?? []).filter((c: any) => c.cost_group === 'SAL')
  const baaItems = (costs ?? []).filter((c: any) => c.cost_group === 'VENDOR')
  const toCostItem = (c: any) => ({ rowRole: c.row_role, harga: Number(c.harga), qty: Number(c.qty), periode: Number(c.periode), kategori: c.kategori, tipe: c.tipe })

  const marginComponents = MARGIN_COMPONENTS_LIST.map((key) => {
    const row = (margins ?? []).find((m: any) => m.component === key)
    return { component: key, percentage: row ? Number(row.percentage) : 0 }
  })

  const pphOn = header?.is_via_salset || fundRows.some((f: any) => f.fund_type === 'CLIENT')

  let result: any = null
  if (header?.cor_method === 'GROSS_DOWN') {
    result = computeGD({
      funds: fundRows.map((f: any) => ({ fundType: f.fund_type, nominal: Number(f.gdv), isZakat: f.is_zakat, techFeeManual: f.tech_fee_manual, manualTechFee: Number(f.manual_tech_fee || 0) })),
      biayaPencairan,
      isViaSalset: header.is_via_salset,
      ngoRatePct: Number(header.ngo_rate_pct),
      biayaSalset: Number(header.biaya_salset),
      pkp,
      pphOn,
      marginEnabled: header.margin_enabled,
      marginMode: header.margin_mode,
      manualMarginPct: Number(header.manual_margin_pct || 0),
      marginComponents,
      salItems: salItems.map(toCostItem),
      baaItems: baaItems.map(toCostItem),
      salsetOnly: header.is_salset_only,
    })
  } else if (header) {
    result = computeGU({
      ngoRatePct: Number(header.ngo_rate_pct || 10),
      isViaSalset: header.is_via_salset,
      pkp,
      biayaPencairan,
      salItems: salItems.map(toCostItem),
      baaItems: baaItems.map(toCostItem),
      marginComponents,
    })
  }

  return { header, vendor, funds: funds ?? [], fundRows, costs: costs ?? [], marginComponents, marginRows: margins ?? [], result, biayaPencairan, pkp }
}

export async function saveDraft(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string

  const { header, result } = await loadCorComputation(admin, docId)
  if (!header) throw new Error('COR Header tidak ditemukan.')

  if (header.cor_method === 'GROSS_DOWN') {
    await admin.from('cor_result').delete().eq('doc_id', docId).eq('cor_tab', 'CLIENT')
    const { error } = await admin.from('cor_result').insert({ doc_id: docId, cor_tab: 'CLIENT', snapshot: result })
    if (error) throw new Error(error.message)
  }

  const { data: doc } = await admin.from('document_pipeline').select('status').eq('id', docId).single()
  if (doc?.status === 'Not Started') {
    await admin.from('document_pipeline').update({ status: 'Drafting', stage: 'In Progress', last_updated: new Date().toISOString() }).eq('id', docId)
  }

  revalidatePath(`/documents/cor/${docId}`)
  revalidatePath('/documents')
}

export async function convertToGrossDown(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string

  const { header, result } = await loadCorComputation(admin, docId)
  if (!header) throw new Error('COR Header tidak ditemukan.')
  if (header.cor_method !== 'GROSS_UP') throw new Error('Cuma COR dengan metode Gross Up yang bisa dikonversi.')

  const { error } = await admin
    .from('cor_header')
    .update({ cor_method: 'GROSS_DOWN', gross_up_snapshot: result, converted_at: new Date().toISOString() })
    .eq('doc_id', docId)
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/cor/${docId}`)
}
