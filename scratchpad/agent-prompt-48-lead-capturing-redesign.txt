import { createAdminClient } from '@/lib/supabase/admin'

export type LeadStats = {
  newLeads: number
  contacted: number
  moved: number
  other: number
  spam: number
  totalAll: number
  totalExcludingSpam: number
  inboundHealthPct: number
  spamPortionPct: number
  stalledOver7Days: number
  duplicatePotentialCount: number
}

export async function getLeadStats(admin: ReturnType<typeof createAdminClient>): Promise<LeadStats> {
  const { data: rows } = await admin.from('lead').select('id, status, email, phone, last_updated')
  const all = rows ?? []

  const newLeads = all.filter((l) => l.status === 'New Leads').length
  const contacted = all.filter((l) => l.status === 'Contacted').length
  const moved = all.filter((l) => l.status === 'Moved').length
  const other = all.filter((l) => l.status === 'Other').length
  const spam = all.filter((l) => l.status === 'Spam').length
  const totalAll = all.length
  const totalExcludingSpam = totalAll - spam

  const inboundHealthPct = totalExcludingSpam > 0 ? Math.round((moved / totalExcludingSpam) * 1000) / 10 : 0
  const spamPortionPct = totalAll > 0 ? Math.round((spam / totalAll) * 1000) / 10 : 0

  const sevenDaysAgoMs = Date.now() - 7 * 24 * 60 * 60 * 1000
  const stalledOver7Days = all.filter(
    (l) => (l.status === 'New Leads' || l.status === 'Contacted') && new Date(l.last_updated).getTime() < sevenDaysAgoMs
  ).length

  // Duplicate detection: any lead whose email OR phone (normalized) is shared
  // with at least one other lead in the whole dataset.
  const emailCounts = new Map<string, number>()
  const phoneCounts = new Map<string, number>()
  for (const l of all) {
    if (l.email) {
      const key = l.email.trim().toLowerCase()
      emailCounts.set(key, (emailCounts.get(key) || 0) + 1)
    }
    if (l.phone) {
      const key = l.phone.replace(/\D/g, '')
      if (key) phoneCounts.set(key, (phoneCounts.get(key) || 0) + 1)
    }
  }
  let duplicatePotentialCount = 0
  for (const l of all) {
    const emailDup = l.email ? (emailCounts.get(l.email.trim().toLowerCase()) || 0) > 1 : false
    const phoneDup = l.phone ? (phoneCounts.get(l.phone.replace(/\D/g, '')) || 0) > 1 : false
    if (emailDup || phoneDup) duplicatePotentialCount++
  }

  return { newLeads, contacted, moved, other, spam, totalAll, totalExcludingSpam, inboundHealthPct, spamPortionPct, stalledOver7Days, duplicatePotentialCount }
}

export async function bulkUpdateLeadStatus(ids: string[], status: string) {
  await requireWriteAccess()
  const allowed = ['Contacted', 'Other', 'Spam']
  if (!allowed.includes(status)) throw new Error('Status massal tidak valid.')
  if (ids.length === 0) return

  const admin = createAdminClient()
  const { data: current } = await admin.from('lead').select('id, status').in('id', ids)
  // Moved leads are locked -- silently skip them rather than failing the whole batch.
  const updatableIds = (current ?? []).filter((l) => l.status !== 'Moved').map((l) => l.id)
  if (updatableIds.length === 0) return

  const { error } = await admin.from('lead').update({ status, last_updated: new Date().toISOString() }).in('id', updatableIds)
  if (error) throw new Error(error.message)

  revalidatePath('/leads')
}

const STATUS_STYLES: Record<string, string> = {
  'New Leads': 'bg-blue-100 text-blue-700',
  Contacted: 'bg-amber-100 text-amber-700',
  Moved: 'bg-green-100 text-green-700',
  Other: 'bg-gray-100 text-gray-700',
  Spam: 'bg-red-100 text-red-700',
}

function daysAgo(dateStr: string): number {
  return Math.max(0, Math.floor((Date.now() - new Date(dateStr).getTime()) / (1000 * 60 * 60 * 24)))
}

export function LeadsTable({ leads, canWrite }: { leads: Lead[]; canWrite: boolean }) {
  const [selected, setSelected] = useState<Set<string>>(new Set())
  const [activeLead, setActiveLead] = useState<Lead | null>(null)
  const [isPending, startTransition] = useTransition()

  function toggleOne(id: string) {
    setSelected((prev) => {
      const next = new Set(prev)
      if (next.has(id)) next.delete(id)
      else next.add(id)
      return next
    })
  }

  function toggleAll() {
    setSelected((prev) => (prev.size === leads.length ? new Set() : new Set(leads.map((l) => l.id))))
  }

  function runBulk(status: string) {
    startTransition(async () => {
      await bulkUpdateLeadStatus(Array.from(selected), status)
      setSelected(new Set())
    })
  }

  return (
    <div>
      {canWrite && selected.size > 0 && (
        <div className="mb-3 flex items-center justify-between rounded-lg bg-gray-900 px-4 py-2 text-sm text-white">
          <span>{selected.size} lead dipilih</span>
          <div className="flex gap-2">
            <button disabled={isPending} onClick={() => runBulk('Contacted')} className="rounded-md bg-white/10 px-3 py-1 hover:bg-white/20 disabled:opacity-50">
              Tandai Contacted
            </button>
            <button disabled={isPending} onClick={() => runBulk('Other')} className="rounded-md bg-white/10 px-3 py-1 hover:bg-white/20 disabled:opacity-50">
              Tandai Other
            </button>
            <button disabled={isPending} onClick={() => runBulk('Spam')} className="rounded-md bg-white/10 px-3 py-1 hover:bg-white/20 disabled:opacity-50">
              Tandai Spam
            </button>
          </div>
        </div>
      )}

      <div className="overflow-hidden rounded-lg border bg-white shadow-sm">
        <table className="w-full text-sm">
          <thead className="bg-gray-50 text-left text-xs font-semibold uppercase tracking-wide text-gray-500">
            <tr>
              <th className="w-10 px-4 py-3">
                {canWrite && (
                  <input type="checkbox" checked={leads.length > 0 && selected.size === leads.length} onChange={toggleAll} />
                )}
              </th>
              <th className="px-4 py-3">Status</th>
              <th className="px-4 py-3">Masuk</th>
              <th className="px-4 py-3">Entity Name</th>
              <th className="px-4 py-3">Kontak</th>
              <th className="px-4 py-3">Prioritas</th>
              <th className="px-4 py-3">Action</th>
            </tr>
          </thead>
          <tbody className="divide-y">
            {leads.map((lead) => (
              <tr key={lead.id} className="cursor-pointer hover:bg-gray-50" onClick={() => setActiveLead(lead)}>
                <td className="px-4 py-3" onClick={(e) => e.stopPropagation()}>
                  {canWrite && <input type="checkbox" checked={selected.has(lead.id)} onChange={() => toggleOne(lead.id)} />}
                </td>
                <td className="px-4 py-3">
                  <span className={`rounded-full px-2 py-0.5 text-xs font-medium ${STATUS_STYLES[lead.status] ?? 'bg-gray-100 text-gray-700'}`}>
                    {lead.status}
                  </span>
                </td>
                <td className="px-4 py-3 whitespace-nowrap text-gray-600">
                  {new Date(lead.submitted_at).toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric' })}
                  <div className="text-xs text-gray-400">{daysAgo(lead.submitted_at)} hari</div>
                </td>
                <td className="px-4 py-3">
                  <div className="font-medium text-gray-900">{lead.entity_name}</div>
                  <div className="text-xs text-gray-500">{lead.pic_name || '-'}</div>
                </td>
                <td className="px-4 py-3 text-gray-600">
                  <div>{lead.phone || '-'}</div>
                  <div className="text-xs text-gray-400">{lead.email || '-'}</div>
                </td>
                <td className="max-w-[200px] truncate px-4 py-3 text-gray-600">{lead.priority_notes || lead.detail_interest || '-'}</td>
                <td className="px-4 py-3" onClick={(e) => e.stopPropagation()}>

function Row({ label, value }: { label: string; value: string }) {

      <dd className="mt-0.5 text-gray-800">{value}</dd>

}

function LeadDrawer({ lead, canWrite, onClose }: { lead: Lead; canWrite: boolean; onClose: () => void }) {
  const [isPending, startTransition] = useTransition()
  const statusOptions = ['New Leads', 'Contacted', 'Other', 'Spam']

  function handleStatusChange(formData: FormData) {
    startTransition(async () => {
      await updateLeadStatus(formData)
      onClose()
    })
  }

  function handleMoveToClient(formData: FormData) {
    startTransition(async () => {
      await moveToClient(formData)
      onClose()
    })
  }

  return (
    <div className="fixed inset-0 z-40 flex justify-end">
      <div className="absolute inset-0 bg-black/30" onClick={onClose} />
      <div className="relative z-10 h-full w-full max-w-md overflow-y-auto bg-white p-6 shadow-xl">
        <div className="mb-4 flex items-start justify-between">
          <div>
            <p className="font-mono text-xs text-gray-400">{lead.inbound_id}</p>
            <h2 className="text-lg font-bold text-gray-900">{lead.entity_name}</h2>
          </div>
          <button onClick={onClose} className="text-xl leading-none text-gray-400 hover:text-gray-700">
            &times;
          </button>
        </div>

        <span className={`mb-4 inline-block rounded-full px-2 py-0.5 text-xs font-medium ${STATUS_STYLES[lead.status] ?? 'bg-gray-100 text-gray-700'}`}>
          {lead.status}
        </span>

        <dl className="mb-6 space-y-3 text-sm">
          <Row label="Entity Type" value={lead.entity_type === 'Other' ? lead.entity_type_other || 'Other' : lead.entity_type || '-'} />

        </dl>

        {canWrite && lead.status !== 'Moved' && (
          <div className="space-y-3 border-t pt-4">
            <form action={handleStatusChange} className="flex items-center gap-2">
              <input type="hidden" name="id" value={lead.id} />

                    {s}

                Move to Client
              </button>
            </form>
          </div>
        )}
        {lead.status === 'Moved' && <p className="border-t pt-4 text-xs italic text-gray-400">Terkunci (sudah jadi Client).</p>}
      </div>
    </div>
  )
}

function StatCard({ label, value, sublabel, tone }: { label: string; value: number | string; sublabel?: string; tone: 'blue' | 'amber' | 'green' | 'gray' | 'red' }) {
  const toneClass: Record<string, string> = {
    blue: 'text-blue-700',
    amber: 'text-amber-700',
    green: 'text-green-700',
    gray: 'text-gray-700',
    red: 'text-red-700',
  }
  return (
    <div className="rounded-lg border bg-white p-4 shadow-sm">
      <p className={`text-xs font-semibold uppercase tracking-wide ${toneClass[tone]}`}>{label}</p>
      <p className="mt-1 text-xl font-bold text-gray-900">{value}</p>
      {sublabel && <p className="mt-0.5 text-xs text-gray-400">{sublabel}</p>}
    </div>
  )
}
