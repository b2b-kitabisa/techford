export async function updateProjectInfo(formData: FormData) {
  const employee = await requireWriteAccess()
  const admin = createAdminClient()
  const id = formData.get('project_id') as string

  const { data: project } = await admin.from('project').select('*').eq('id', id).single()
  if (!project) throw new Error('Project tidak ditemukan.')

  const otherNotes = (formData.get('other_notes') as string) || null

  if (project.stage === 'Won') {
    // Won-locked: cuma other_notes yang boleh diubah lewat form ini.
    const { error } = await admin.from('project').update({ other_notes: otherNotes, last_updated: new Date().toISOString() }).eq('id', id)
    if (error) throw new Error(error.message)
    revalidatePath(`/projects/${id}`)
    revalidatePath('/projects')
    return
  }

  const services = formData.getAll('services') as string[]
  const issues = formData.getAll('issues') as string[]
  // is_retainer sekali true, tidak bisa dimatikan lagi.
  const isRetainer = project.is_retainer || formData.get('is_retainer') === 'on'

  const { error } = await admin.from('project').update({
    project_name: ((formData.get('project_name') as string) || '').trim(),
    consultant_name_fallback: (formData.get('consultant_name_fallback') as string) || null,
    services,
    issues,
    program_type: (formData.get('program_type') as string) || null,
    program_category: (formData.get('program_category') as string) || null,
    program_name: (formData.get('program_name') as string) || null,
    is_retainer: isRetainer,
    other_notes: otherNotes,
    last_updated: new Date().toISOString(),
  }).eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath(`/projects/${id}`)
  revalidatePath('/projects')
}

function rupiah(n: number) {
  return 'Rp ' + Math.round(n).toLocaleString('id-ID')
}
