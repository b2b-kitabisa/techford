The text/date search bar stays server-side as before since it needs a real database query.

function CopyButton({ text }: { text: string }) {
  const [copied, setCopied] = useState(false)

function LeadDrawer({

      if (e.key === 'ArrowLeft' && index > 0) onNavigate(index - 1)

              <span className={`rounded-full px-2 py-0.5 text-xs font-medium ${STATUS_STYLES[lead.status] ?? 'bg-white/10 text-white'}`}>{lead.status}</span>
              <span className="font-mono text-xs text-gray-400">{lead.inbound_id}</span>
            </div>
            <div className="flex items-center gap-1">
              <button
                disabled={index <= 0}
                onClick={() => onNavigate(index - 1)}
                className="rounded-md px-2 py-1 text-gray-300 hover:bg-white/10 disabled:opacity-30"
              >
                &lsaquo;
              </button>
              <span className="text-xs text-gray-400">
                {index + 1}/{total}
              </span>
              <button
                disabled={index >= total - 1}
                onClick={() => onNavigate(index + 1)}
                className="rounded-md px-2 py-1 text-gray-300 hover:bg-white/10 disabled:opacity-30"
              >
                &rsaquo;
              </button>
              <button onClick={onClose} className="ml-1 rounded-md px-2 py-1 text-lg leading-none text-gray-300 hover:bg-white/10">
                &times;
              </button>
            </div>
          </div>

          <h2 className="mt-3 text-lg font-bold">{lead.entity_name}</h2>
          <span className="mt-1 inline-block rounded-full bg-white/10 px-2 py-0.5 text-xs font-medium uppercase tracking-wide text-gray-200">
            {lead.entity_type === 'Other' ? lead.entity_type_other || 'Other' : lead.entity_type || '-'}
          </span>

          {canWrite && lead.status !== 'Moved' && (
            <div className="mt-4 flex items-center justify-between gap-2">
              <form action={handleStatusChange} className="flex items-center gap-2">
                <input type="hidden" name="id" value={lead.id} />
                <select
                  name="status"
                  defaultValue={lead.status}
                  className="rounded-md border border-gray-700 bg-gray-800 px-2 py-1.5 text-xs text-white"
                >
                  {STATUS_OPTIONS.map((s) => (
                    <option key={s} value={s}>
                      {s}
                    </option>
                  ))}
                </select>
                <button
                  type="submit"
                  disabled={isPending}
                  className="inline-flex items-center gap-1.5 rounded-md bg-blue-600 px-3 py-1.5 text-xs font-medium text-white hover:bg-blue-500 disabled:cursor-not-allowed disabled:opacity-50"
                >
                  {isPending && <Spinner />} Update
                </button>
              </form>
              <form action={handleMoveToClient}>
                <input type="hidden" name="lead_id" value={lead.id} />
                <button
                  type="submit"
                  disabled={isPending}
                  className="inline-flex items-center gap-1.5 rounded-md bg-indigo-600 px-3 py-1.5 text-xs font-medium text-white hover:bg-indigo-500 disabled:cursor-not-allowed disabled:opacity-50"
                >
                  {isPending && <Spinner />} &raquo; Move to Client
                </button>
              </form>
            </div>
          )}
          {lead.status === 'Moved' && <p className="mt-4 text-xs italic text-gray-400">Terkunci (sudah jadi Client).</p>}
        </div>

        <div className="flex-1 overflow-y-auto p-5">
          <div className="mb-5 flex items-center gap-3">
            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-blue-500 text-sm font-semibold text-white">
              {initials(lead.pic_name)}
            </div>
            <div className="min-w-0 flex-1">
              <p className="truncate text-sm font-semibold text-gray-900">{lead.pic_name || '-'}</p>
              {lead.phone && (
                <div className="flex items-center gap-1">
                  <a href={`tel:${lead.phone}`} className="truncate text-xs text-blue-600 hover:underline">
                    {formatPhone(lead.phone)}
                  </a>
                  <CopyButton text={formatPhone(lead.phone)} />
                </div>
              )}
              {lead.email && (
                <div className="flex items-center gap-1">
                  <a href={`mailto:${lead.email}`} className="truncate text-xs text-blue-600 hover:underline">
                    {lead.email}
                  </a>
                  <CopyButton text={lead.email} />
                </div>
              )}
            </div>
          </div>

          {lead.duplicates.length > 0 && (
            <div className="mb-4">
              <DuplicateSection duplicates={lead.duplicates} onDone={() => router.refresh()} />
            </div>
          )}

          <div className="border-t pt-4">
            <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-gray-400">Interest</p>
            <div className="mb-3">
              <p className="text-xs font-semibold text-gray-500">Kebutuhan</p>
              <p className="mt-0.5 text-sm text-gray-800">{lead.detail_interest || '-'}</p>
            </div>
            <div>
              <p className="text-xs font-semibold text-gray-500">Prioritas</p>
              <p className="mt-0.5 text-sm text-gray-800">{lead.priority_notes || '-'}</p>
            </div>
          </div>

          {trafficTags.length > 0 && (
            <div className="mt-4 border-t pt-4">
              <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-gray-400">Sumber Trafik</p>
              <div className="flex flex-wrap gap-2">
                {trafficTags.map((t) => (
                  <span key={t.label} className="rounded-full border border-gray-200 bg-gray-50 px-3 py-1 text-xs text-gray-600">
                    {t.label} <span className="font-semibold text-gray-800">{t.value}</span>
                  </span>
                ))}
              </div>
            </div>
          )}

          <div className="mt-4 border-t pt-4">
            <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-gray-400">Other Notes</p>
            <p className="text-sm text-gray-600">{lead.other_notes || 'Belum ada catatan.'}</p>
          </div>
        </div>
      </div>
    </div>
  )
}

function DuplicateSection({ duplicates, onDone }: { duplicates: DuplicateRef[]; onDone: () => void }) {
  const [selected, setSelected] = useState<Set<string>>(new Set(duplicates.map((d) => d.id)))
  const [bulkStatus, setBulkStatus] = useState('Spam')
  const [isPending, startTransition] = useTransition()

  function toggle(id: string) {
    setSelected((prev) => {
      const next = new Set(prev)
      if (next.has(id)) next.delete(id)
      else next.add(id)
      return next
    })
  }

  function applyBulk() {
    startTransition(async () => {
      await bulkUpdateLeadStatus(Array.from(selected), bulkStatus)
      onDone()
    })
  }

  return (
    <details className="rounded-md border border-orange-200 bg-orange-50" open>
      <summary className="cursor-pointer px-3 py-2 text-xs font-semibold uppercase tracking-wide text-orange-700">
        Potensi Duplikat ({duplicates.length})
      </summary>
      <div className="space-y-2 border-t border-orange-200 px-3 py-2">
        {duplicates.map((d) => (
          <label key={d.id} className="flex items-center gap-2 text-xs text-gray-700">
            <input type="checkbox" checked={selected.has(d.id)} onChange={() => toggle(d.id)} />
            <span className="flex-1 truncate">
              {d.entity_name} <span className="text-gray-400">({d.inbound_id})</span>
            </span>
            <span className={`rounded-full px-1.5 py-0.5 text-[10px] font-medium ${STATUS_STYLES[d.status] ?? 'bg-gray-100 text-gray-700'}`}>{d.status}</span>
            <span className="text-[10px] text-gray-400">via {d.matched_by === 'email' ? 'Email' : 'Phone'}</span>
          </label>
        ))}
        <div className="flex items-center gap-2 pt-2">
          <select value={bulkStatus} onChange={(e) => setBulkStatus(e.target.value)} className="rounded-md border-gray-300 text-xs">
            {STATUS_OPTIONS.map((s) => (
              <option key={s} value={s}>
                {s}
              </option>
            ))}
          </select>
          <button
            onClick={applyBulk}
            disabled={isPending || selected.size === 0}
            className="inline-flex items-center gap-1.5 rounded-md bg-orange-600 px-3 py-1.5 text-xs font-medium text-white hover:bg-orange-500 disabled:cursor-not-allowed disabled:opacity-50"
          >
            {isPending && <Spinner />} Update {selected.size} Data Terpilih
          </button>
        </div>
      </div>
    </details>
  )
}
