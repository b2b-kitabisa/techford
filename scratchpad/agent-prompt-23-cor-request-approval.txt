import nodemailer from 'nodemailer'

export async function sendEmail({ to, subject, html }: { to: string; subject: string; html: string }) {
  const transporter = nodemailer.createTransport({
    host: process.env.SMTP_HOST,
    port: Number(process.env.SMTP_PORT || 587),
    secure: false,
    auth: { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS },
  })
  await transporter.sendMail({
    from: process.env.EMAIL_FROM || process.env.SMTP_USER,
    to,
    subject,
    html,
  })
}

import { randomUUID } from 'crypto'
import { evaluateMarginGuard } from '@/lib/cor-calc'
import { loadCorComputation } from '@/lib/cor-load'
import { sendEmail } from '@/lib/send-email'

export async function requestApproval(formData: FormData) {
  const employee = await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string
  const approverId = formData.get('approver_id') as string
  const marginAckNote = ((formData.get('margin_ack_note') as string) || '').trim()

  const { header, result } = await loadCorComputation(admin, docId)
  if (!header) throw new Error('COR Header tidak ditemukan.')
  if (header.cor_method !== 'GROSS_DOWN') throw new Error('Cuma metode Gross Down yang bisa diajukan approval. Convert dulu.')

  const { data: approver } = await admin.from('employee').select('*').eq('id', approverId).single()
  if (!approver || approver.role !== 'Head of B2B') throw new Error('Approver harus employee dengan role Head of B2B.')

  if (!header.is_salset_only) {
    const guard = evaluateMarginGuard(result.totalMgnFrac ?? 0, result.pmPct ?? 0)
    if (guard.below && !marginAckNote) {
      throw new Error(`Margin aktual (${guard.actualPct}%) di bawah panduan (${guard.planPct}%) -- wajib isi alasan sebelum lanjut.`)
    }
  }

  const token = randomUUID()
  const expiresAt = new Date(Date.now() + 14 * 24 * 60 * 60 * 1000).toISOString()

  const { error: headerErr } = await admin.from('cor_header').update({
    approval_token: token,
    approval_expires_at: expiresAt,
    approval_requested_to: approver.email,
    approval_requested_name: approver.name,
    approval_requested_at: new Date().toISOString(),
    approval_resolved_at: null,
    rejection_note: null,
  }).eq('doc_id', docId)
  if (headerErr) throw new Error(headerErr.message)

  const { error: docErr } = await admin
    .from('document_pipeline')
    .update({ status: 'Waiting Approval', stage: 'In Progress', last_updated: new Date().toISOString() })
    .eq('id', docId)
  if (docErr) throw new Error(docErr.message)

  await admin.from('document_activity').insert({
    doc_id: docId,
    activity_type: 'APPROVAL_REQUESTED',
    round_no: 1,
    actor_name: employee.name,
    actor_email: employee.email,
  })

  const { data: doc } = await admin.from('document_pipeline').select('doc_id').eq('id', docId).single()
  const baseUrl = process.env.NEXT_PUBLIC_APP_URL || ''
  const approveUrl = `${baseUrl}/approve/cor/${token}?action=approve`
  const rejectUrl = `${baseUrl}/approve/cor/${token}?action=reject`
  const marginNoteHtml = marginAckNote
    ? `<p style="color:#b91c1c;font-weight:bold;">Catatan margin di bawah panduan: ${marginAckNote}</p>`
    : ''

  await sendEmail({
    to: approver.email,
    subject: `Approval COR ${doc?.doc_id} diperlukan`,
    html: `
      ${marginNoteHtml}
      <p>Dokumen COR <strong>${doc?.doc_id}</strong> menunggu approval Anda.</p>
      <p><a href="${approveUrl}">Approve</a> &nbsp;|&nbsp; <a href="${rejectUrl}">Reject</a></p>
      <p style="color:#888;font-size:12px;">Link berlaku 14 hari, tanpa perlu login.</p>
    `,
  })

  revalidatePath(`/documents/cor/${docId}`)
  revalidatePath('/documents')
}
