function pct(n: number, total: number): number {
  return total > 0 ? Math.round((n / total) * 1000) / 10 : 0
}

function StatCard({ label, value, sublabel, tone }: { label: string; value: number | string; sublabel?: string; tone: 'blue' | 'amber' | 'green' | 'gray' | 'red' }) {
  const toneClass: Record<string, string> = {
    blue: 'text-blue-700',
    amber: 'text-amber-700',
    green: 'text-green-700',
    gray: 'text-gray-700',
    red: 'text-red-700',
  }
  return (
    <div className="rounded-lg border bg-white p-3 shadow-sm">
      <p className={`text-xs font-semibold uppercase tracking-wide ${toneClass[tone]}`}>{label}</p>
      <p className="mt-1 text-lg font-bold text-gray-900">{value}</p>
      {sublabel && <p className="mt-0.5 text-xs text-gray-400">{sublabel}</p>}
    </div>
  )
}

export default async function LeadsPage({
  searchParams,
}: {
  searchParams: Promise<{ q?: string; status?: string; from?: string; to?: string }>
}) {
  const sp = await searchParams
  const employee = await getCurrentEmployee()
  if (!employee) redirect('/login')

  const admin = createAdminClient()
  const stats = await getLeadStats(admin)

  let query = admin.from('lead').select('*').order('submitted_at', { ascending: false })
  if (sp.status) query = query.eq('status', sp.status)
  if (sp.from) query = query.gte('submitted_at', sp.from)
  if (sp.to) query = query.lte('submitted_at', `${sp.to}T23:59:59`)
  if (sp.q) {
    const q = `%${sp.q}%`
    query = query.or(`entity_name.ilike.${q},pic_name.ilike.${q},email.ilike.${q},phone.ilike.${q}`)
  }
  const { data: leads } = await query

  const canWrite = employee.role !== 'Operation'

  return (
    <AppShell employeeName={employee.name} employeeRole={employee.role} breadcrumb={{ section: 'Sales Module', page: 'Lead Capturing' }}>
      <div className="mb-4 grid grid-cols-4 grid-rows-2 gap-3">
        <StatCard label="New Leads" value={stats.newLeads} sublabel={`${pct(stats.newLeads, stats.totalAll)}% dari Total Leads`} tone="blue" />
        <StatCard label="Contacted" value={stats.contacted} sublabel={`${pct(stats.contacted, stats.totalAll)}% dari Total Leads`} tone="amber" />
        <StatCard label="Moved" value={stats.moved} sublabel={`${pct(stats.moved, stats.totalAll)}% dari Total Leads`} tone="green" />

            </div>
            <div className="flex items-center justify-between">
              <span className="text-blue-600">Porsi spam</span>
              <span className="font-semibold text-blue-900">{stats.spamPortionPct}%</span>
            </div>
            <div className="flex items-center justify-between">
              <span className="text-red-500">Potensi duplikat</span>

'use client'

import { useEffect, useState, useTransition } from 'react'
import { updateLeadStatus, moveToClient, bulkUpdateLeadStatus } from './actions'

type Lead = {
  id: string
  inbound_id: string
  status: string
  submitted_at: string
  entity_name: string
  entity_type: string | null
  entity_type_other: string | null
  pic_name: string | null
  email: string | null
  phone: string | null
  detail_interest: string | null
  priority_notes: string | null
  utm_source: string | null
  utm_medium: string | null
  utm_campaign: string | null
  other_notes: string | null
  last_updated: string
}

const STATUS_STYLES: Record<string, string> = {
  'New Leads': 'bg-blue-100 text-blue-700',
  Contacted: 'bg-amber-100 text-amber-700',
  Moved: 'bg-green-100 text-green-700',
  Other: 'bg-gray-100 text-gray-700',
  Spam: 'bg-red-100 text-red-700',
}

const STATUS_OPTIONS = ['New Leads', 'Contacted', 'Other', 'Spam']

function daysAgo(dateStr: string): number {
  return Math.max(0, Math.floor((Date.now() - new Date(dateStr).getTime()) / (1000 * 60 * 60 * 24)))
}

function initials(name: string | null): string {
  if (!name) return '?'
  const parts = name.trim().split(/\s+/).filter(Boolean)
  return parts.slice(0, 2).map((p) => p[0]?.toUpperCase() || '').join('') || '?'
}

function copyText(text: string) {
  if (navigator.clipboard) navigator.clipboard.writeText(text).catch(() => {})
}

export function LeadsTable({ leads, canWrite }: { leads: Lead[]; canWrite: boolean }) {
  const [selected, setSelected] = useState<Set<string>>(new Set())
  const [activeIndex, setActiveIndex] = useState<number | null>(null)
  const [isPending, startTransition] = useTransition()

  function toggleOne(id: string) {
    setSelected((prev) => {
      const next = new Set(prev)
      if (next.has(id)) next.delete(id)
      else next.add(id)
      return next
    })
  }

  function toggleAll() {
    setSelected((prev) => (prev.size === leads.length ? new Set() : new Set(leads.map((l) => l.id))))
  }

  function runBulk(status: string) {
    startTransition(async () => {
      await bulkUpdateLeadStatus(Array.from(selected), status)
      setSelected(new Set())
    })
  }

  return (
    <div>
      {canWrite && selected.size > 0 && (
        <div className="mb-3 flex items-center justify-between rounded-lg bg-gray-900 px-4 py-2 text-sm text-white">
          <span>{selected.size} lead dipilih</span>
          <div className="flex gap-2">
            <button disabled={isPending} onClick={() => runBulk('Contacted')} className="rounded-md bg-white/10 px-3 py-1 hover:bg-white/20 disabled:opacity-50">
              Tandai Contacted
            </button>
            <button disabled={isPending} onClick={() => runBulk('Other')} className="rounded-md bg-white/10 px-3 py-1 hover:bg-white/20 disabled:opacity-50">
              Tandai Other
            </button>
            <button disabled={isPending} onClick={() => runBulk('Spam')} className="rounded-md bg-white/10 px-3 py-1 hover:bg-white/20 disabled:opacity-50">
              Tandai Spam
            </button>
          </div>
        </div>
      )}

      <div className="overflow-hidden rounded-lg border bg-white shadow-sm">
        <table className="w-full text-sm">
          <thead className="bg-gray-50 text-left text-xs font-semibold uppercase tracking-wide text-gray-500">
            <tr>
              <th className="w-10 px-4 py-2">
                {canWrite && (
                  <input type="checkbox" checked={leads.length > 0 && selected.size === leads.length} onChange={toggleAll} />
                )}
              </th>
              <th className="px-4 py-2">Status</th>
              <th className="px-4 py-2">Masuk</th>
              <th className="px-4 py-2">Entity Name</th>
              <th className="px-4 py-2">Kontak</th>
              <th className="px-4 py-2">Prioritas</th>
              <th className="px-4 py-2">Action</th>
            </tr>
          </thead>
          <tbody className="divide-y">
            {leads.map((lead, index) => (
              <tr key={lead.id} className="cursor-pointer hover:bg-gray-50" onClick={() => setActiveIndex(index)}>
                <td className="px-4 py-2" onClick={(e) => e.stopPropagation()}>
                  {canWrite && <input type="checkbox" checked={selected.has(lead.id)} onChange={() => toggleOne(lead.id)} />}
                </td>
                <td className="px-4 py-2">
                  <span className={`rounded-full px-2 py-0.5 text-xs font-medium ${STATUS_STYLES[lead.status] ?? 'bg-gray-100 text-gray-700'}`}>
                    {lead.status}
                  </span>
                </td>
                <td className="px-4 py-2 whitespace-nowrap text-gray-600">
                  {new Date(lead.submitted_at).toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric' })}
                  <div className="text-xs text-gray-400">{daysAgo(lead.submitted_at)} hari</div>
                </td>
                <td className="px-4 py-2">
                  <div className="font-medium text-gray-900">{lead.entity_name}</div>
                  <div className="text-xs text-gray-500">{lead.pic_name || '-'}</div>
                </td>
                <td className="px-4 py-2 text-gray-600">
                  <div>{lead.phone || '-'}</div>
                  <div className="text-xs text-gray-400">{lead.email || '-'}</div>
                </td>
                <td className="max-w-[200px] truncate px-4 py-2 text-gray-600">{lead.priority_notes || lead.detail_interest || '-'}</td>
                <td className="px-4 py-2" onClick={(e) => e.stopPropagation()}>
                  <button onClick={() => setActiveIndex(index)} className="rounded-full border px-2 py-1 text-xs text-gray-500 hover:bg-gray-100">
                    &rsaquo;
                  </button>
                </td>
              </tr>
            ))}
            {leads.length === 0 && (
              <tr>
                <td colSpan={7} className="px-4 py-8 text-center text-gray-400">
                  Tidak ada Lead yang cocok dengan filter.
                </td>
              </tr>
            )}
          </tbody>
        </table>
      </div>

      {activeIndex !== null && leads[activeIndex] && (
        <LeadDrawer
          lead={leads[activeIndex]}
          index={activeIndex}
          total={leads.length}
          canWrite={canWrite}
          onClose={() => setActiveIndex(null)}
          onNavigate={(i) => setActiveIndex(i)}
        />
      )}
    </div>
  )
}

function LeadDrawer({
  lead,
  index,
  total,
  canWrite,
  onClose,
  onNavigate,
}: {
  lead: Lead
  index: number
  total: number
  canWrite: boolean
  onClose: () => void
  onNavigate: (i: number) => void
}) {
  const [isPending, startTransition] = useTransition()

  // Only the drawer scrolls while it's open -- the page/table behind it must
  // not also scroll (that was the reported bug).
  useEffect(() => {
    document.body.style.overflow = 'hidden'
    return () => {
      document.body.style.overflow = ''
    }
  }, [])

  function handleStatusChange(formData: FormData) {
    startTransition(async () => {
      await updateLeadStatus(formData)
    })
  }

  function handleMoveToClient(formData: FormData) {
    startTransition(async () => {
      await moveToClient(formData)
      onClose()
    })
  }

  const trafficTags = [
    lead.utm_source && { label: 'source', value: lead.utm_source },
    lead.utm_medium && { label: 'medium', value: lead.utm_medium },
    lead.utm_campaign && { label: 'campaign', value: lead.utm_campaign },
  ].filter(Boolean) as { label: string; value: string }[]

  return (
    <div className="fixed inset-0 z-40 flex justify-end">
      <div className="absolute inset-0 bg-black/30" onClick={onClose} />
      <div className="relative z-10 flex h-full w-full max-w-md flex-col bg-white shadow-xl">
        {/* Dark header block */}
        <div className="shrink-0 bg-gray-900 p-5 text-white">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-2">
              <span className={`rounded-full px-2 py-0.5 text-xs font-medium ${STATUS_STYLES[lead.status] ?? 'bg-white/10 text-white'}`}>{lead.status}</span>
              <span className="font-mono text-xs text-gray-400">{lead.inbound_id}</span>
            </div>
            <div className="flex items-center gap-1">
              <button
                disabled={index <= 0}
                onClick={() => onNavigate(index - 1)}
                className="rounded-md px-2 py-1 text-gray-300 hover:bg-white/10 disabled:opacity-30"
              >
                &lsaquo;
              </button>
              <span className="text-xs text-gray-400">
                {index + 1}/{total}
              </span>
              <button
                disabled={index >= total - 1}
                onClick={() => onNavigate(index + 1)}
                className="rounded-md px-2 py-1 text-gray-300 hover:bg-white/10 disabled:opacity-30"
              >
                &rsaquo;
              </button>
              <button onClick={onClose} className="ml-1 rounded-md px-2 py-1 text-lg leading-none text-gray-300 hover:bg-white/10">
                &times;
              </button>
            </div>
          </div>

          <h2 className="mt-3 text-lg font-bold">{lead.entity_name}</h2>
          <span className="mt-1 inline-block rounded-full bg-white/10 px-2 py-0.5 text-xs font-medium uppercase tracking-wide text-gray-200">
            {lead.entity_type === 'Other' ? lead.entity_type_other || 'Other' : lead.entity_type || '-'}
          </span>

          {canWrite && lead.status !== 'Moved' && (
            <div className="mt-4 flex items-center justify-between gap-2">
              <form action={handleStatusChange} className="flex items-center gap-2">
                <input type="hidden" name="id" value={lead.id} />
                <select
                  name="status"
                  defaultValue={lead.status}
                  className="rounded-md border border-gray-700 bg-gray-800 px-2 py-1.5 text-xs text-white"
                >
                  {STATUS_OPTIONS.map((s) => (
                    <option key={s} value={s}>
                      {s}
                    </option>
                  ))}
                </select>
                <button type="submit" disabled={isPending} className="rounded-md bg-blue-600 px-3 py-1.5 text-xs font-medium text-white hover:bg-blue-500 disabled:opacity-50">
                  Update
                </button>
              </form>
              <form action={handleMoveToClient}>
                <input type="hidden" name="lead_id" value={lead.id} />
                <button
                  type="submit"
                  disabled={isPending}
                  className="rounded-md bg-indigo-600 px-3 py-1.5 text-xs font-medium text-white hover:bg-indigo-500 disabled:opacity-50"
                >
                  &raquo; Move to Client
                </button>
              </form>
            </div>
          )}
          {lead.status === 'Moved' && <p className="mt-4 text-xs italic text-gray-400">Terkunci (sudah jadi Client).</p>}
        </div>

        {/* Scrollable body -- this is the ONLY element that scrolls */}
        <div className="flex-1 overflow-y-auto p-5">

            <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-blue-500 text-sm font-semibold text-white">

            </div>

          {trafficTags.length > 0 && (
            <div className="border-t pt-4 mt-4">
              <p className="mb-2 text-xs font-semibold uppercase tracking-wide text-gray-400">Sumber Trafik</p>

                    {t.label} <span className="font-semibold text-gray-800">{t.value}</span>
                  </span>
