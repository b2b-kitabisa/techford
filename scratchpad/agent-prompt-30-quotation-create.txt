'use server'

import { redirect } from 'next/navigation'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'
import { romanMonth } from '@/lib/roman-month'

const ENTITY_LABELS: Record<string, string> = {
  YKB: 'YKB (Yayasan Kita Bisa)',
  KAI: 'PT KAI (PT Kolaborasi Aksi Indonesia)',
}

export async function createQuotationDocument(formData: FormData) {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Operation') throw new Error('Role Operation tidak bisa membuat Quotation.')

  const admin = createAdminClient()
  const projectId = formData.get('project_id') as string
  const entityCode = formData.get('entity_code') as string
  const picClientId = (formData.get('pic_client_id') as string) || null
  if (!projectId) throw new Error('Project wajib dipilih.')
  if (!ENTITY_LABELS[entityCode]) throw new Error('Entity tidak valid.')

  const { data: project } = await admin.from('project').select('client_id').eq('id', projectId).single()
  if (!project) throw new Error('Project tidak ditemukan.')
  const { data: client } = await admin.from('client').select('client_id').eq('id', project.client_id).single()

  const year = new Date().getFullYear() % 100
  const { data: docSeq, error: docSeqErr } = await admin.rpc('next_sequence', { seq_key: `DOC_${year}` })
  if (docSeqErr) throw new Error(docSeqErr.message)
  const docId = `DOC${year}-${String(docSeq).padStart(5, '0')}`

  const { data: doc, error: docErr } = await admin
    .from('document_pipeline')
    .insert({
      doc_id: docId,
      project_id: projectId,
      document_type: 'QUOTATION',
      entity: ENTITY_LABELS[entityCode],
      status: 'Not Started',
      stage: 'New Request',
      requested_by: employee.name,
    })
    .select()
    .single()
  if (docErr || !doc) {
    if (docErr?.message.includes('uniq_quotation_per_project_entity')) {
      throw new Error('Project ini sudah punya Quotation untuk entity yang sama.')
    }
    throw new Error(docErr?.message || 'Gagal membuat dokumen.')
  }

  // Combined YKB+KAI sequence per year, formatted QO/0001/VIII/2026/KAI/CL26-00001
  const fullYear = new Date().getFullYear()
  const { data: qSeq, error: qSeqErr } = await admin.rpc('next_sequence', { seq_key: `QUOTATION_${year}` })
  if (qSeqErr) throw new Error(qSeqErr.message)
  const quotationNumber = `QO/${String(qSeq).padStart(4, '0')}/${romanMonth(new Date().getMonth())}/${fullYear}/${entityCode}/${client?.client_id || '-'}`

  let picSnapshot = { pic_name: null as string | null, pic_email: null as string | null, pic_phone: null as string | null, pic_title: null as string | null }
  if (picClientId) {
    const { data: pic } = await admin.from('pic_client').select('*').eq('id', picClientId).single()
    if (pic) picSnapshot = { pic_name: pic.pic_name, pic_email: pic.email, pic_phone: pic.phone, pic_title: pic.title }
  }

  const { error: headerErr } = await admin.from('quotation_header').insert({
    doc_id: doc.id,
    entity_code: entityCode,
    language: 'ID',
    quotation_number: quotationNumber,
    valid_days: 14,
    pic_client_id: picClientId,
    ...picSnapshot,
    agency_fee_rate: entityCode === 'KAI' ? 5 : null,
    created_by: employee.name,
  })
  if (headerErr) throw new Error(headerErr.message)

  redirect(`/documents/quotation/${doc.id}`)
}

'use server'

import { revalidatePath } from 'next/cache'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'
import { sendEmail } from '@/lib/send-email'
import { randomUUID } from 'crypto'

async function requireWriteAccess() {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Operation') throw new Error('Role Operation tidak bisa mengubah Quotation.')
  return employee
}

export async function saveQuotationDraft(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string

  const { error } = await admin.from('quotation_header').update({
    head_name: (formData.get('head_name') as string) || null,
    title_name: (formData.get('title_name') as string) || null,
    service_name: (formData.get('service_name') as string) || null,
    valid_days: Number(formData.get('valid_days') || 14),
    agency_fee_rate: formData.get('agency_fee_rate') ? Number(formData.get('agency_fee_rate')) : null,
    hide_agency_fee: formData.get('hide_agency_fee') === 'on',
    single_box_price: formData.get('single_box_price') === 'on',
    first_statement: (formData.get('first_statement') as string) || null,
    important_remarks: (formData.get('important_remarks') as string) || null,
    last_updated: new Date().toISOString(),
  }).eq('doc_id', docId)
  if (error) throw new Error(error.message)

  const { data: doc } = await admin.from('document_pipeline').select('status').eq('id', docId).single()
  if (doc?.status === 'Not Started') {
    await admin.from('document_pipeline').update({ status: 'Drafting', stage: 'In Progress', last_updated: new Date().toISOString() }).eq('id', docId)
  }

  revalidatePath(`/documents/quotation/${docId}`)
  revalidatePath('/documents')
}

export async function addQuotationItem(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string
  const itemLabel = ((formData.get('item_label') as string) || '').trim()
  const value = Number(formData.get('value') || 0)
  if (!itemLabel) throw new Error('Item Label wajib diisi.')

  const { error } = await admin.from('quotation_item').insert({
    doc_id: docId,
    category_label: (formData.get('category_label') as string) || null,
    row_role: 'PRICE',
    item_label: itemLabel,
    value,
    qty: Number(formData.get('qty') || 1),
    remarks_detail: (formData.get('remarks_detail') as string) || null,
  })
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/quotation/${docId}`)
}

export async function removeQuotationItem(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const id = formData.get('id') as string
  const docId = formData.get('doc_id') as string

  const { error } = await admin.from('quotation_item').delete().eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath(`/documents/quotation/${docId}`)
}

export async function requestQuotationApproval(formData: FormData) {
  const employee = await requireWriteAccess()
  const admin = createAdminClient()
  const docId = formData.get('doc_id') as string
  const approverId = formData.get('approver_id') as string

  const { data: approver } = await admin.from('employee').select('*').eq('id', approverId).single()
  if (!approver || approver.role !== 'Head of B2B') throw new Error('Approver harus employee dengan role Head of B2B.')

  const token = randomUUID()
  const expiresAt = new Date(Date.now() + 14 * 24 * 60 * 60 * 1000).toISOString()

  const { error: headerErr } = await admin.from('quotation_header').update({
    approval_token: token,
    approval_expires_at: expiresAt,
    approval_requested_to: approver.email,
    approval_requested_name: approver.name,
    approval_requested_at: new Date().toISOString(),
    approval_resolved_at: null,
    rejection_note: null,
  }).eq('doc_id', docId)
  if (headerErr) throw new Error(headerErr.message)

  await admin.from('document_pipeline').update({ status: 'Waiting Approval', stage: 'In Progress', last_updated: new Date().toISOString() }).eq('id', docId)

  await admin.from('document_activity').insert({
    doc_id: docId,
    activity_type: 'APPROVAL_REQUESTED',
    round_no: 1,
    actor_name: employee.name,
    actor_email: employee.email,
  })

  const { data: doc } = await admin.from('document_pipeline').select('doc_id').eq('id', docId).single()
  const { data: header } = await admin.from('quotation_header').select('quotation_number').eq('doc_id', docId).single()
  const baseUrl = process.env.NEXT_PUBLIC_APP_URL || ''
  const approveUrl = `${baseUrl}/approve/quotation/${token}?action=approve`
  const rejectUrl = `${baseUrl}/approve/quotation/${token}?action=reject`

  await sendEmail({
    to: approver.email,
    subject: `Approval Quotation ${doc?.doc_id} (${header?.quotation_number}) diperlukan`,
    html: `
      <p>Dokumen Quotation <strong>${doc?.doc_id}</strong> (${header?.quotation_number}) menunggu approval Anda.</p>
      <p><a href="${approveUrl}">Approve</a> &nbsp;|&nbsp; <a href="${rejectUrl}">Reject</a></p>
      <p style="color:#888;font-size:12px;">Link berlaku 14 hari, tanpa perlu login.</p>
    `,
  })

  revalidatePath(`/documents/quotation/${docId}`)
  revalidatePath('/documents')
}

  return 'Rp ' + Math.round(n).toLocaleString('id-ID')
}
