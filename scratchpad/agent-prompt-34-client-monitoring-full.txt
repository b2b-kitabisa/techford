'use server'

import { revalidatePath } from 'next/cache'
import { redirect } from 'next/navigation'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'

async function requireWriteAccess() {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role === 'Operation') throw new Error('Role Operation cuma bisa lihat Client Monitoring, tidak bisa mengubah.')
  return employee
}

export async function createClient(formData: FormData) {
  const employee = await requireWriteAccess()
  const admin = createAdminClient()

  const brandName = ((formData.get('brand_name') as string) || '').trim()
  const clientSource = formData.get('client_source') as string
  if (!brandName) throw new Error('Brand Name wajib diisi.')
  if (!clientSource) throw new Error('Client Source wajib dipilih.')

  const year = new Date().getFullYear() % 100
  const { data: seqValue, error: seqErr } = await admin.rpc('next_sequence', { seq_key: `CLIENT_${year}` })
  if (seqErr) throw new Error(seqErr.message)
  const clientId = `CL${year}-${String(seqValue).padStart(5, '0')}`

  const { data: client, error } = await admin
    .from('client')
    .insert({
      client_id: clientId,
      brand_name: brandName.toUpperCase(),
      entity_name: ((formData.get('entity_name') as string) || '').toUpperCase(),
      entity_type: (formData.get('entity_type') as string) || null,
      head_office: (formData.get('head_office') as string) || null,
      website: (formData.get('website') as string) || null,
      industry: (formData.get('industry') as string) || null,
      client_source: clientSource,
      is_from_lead: false,
      other_notes: (formData.get('other_notes') as string) || null,
      created_by: employee.name,
    })
    .select()
    .single()
  if (error || !client) throw new Error(error?.message || 'Gagal membuat Client.')

  redirect(`/clients/${client.id}`)
}

export async function updateClient(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const id = formData.get('id') as string

  const { data: existing } = await admin.from('client').select('is_from_lead, client_source').eq('id', id).single()
  if (!existing) throw new Error('Client tidak ditemukan.')

  const patch: Record<string, unknown> = {
    brand_name: ((formData.get('brand_name') as string) || '').toUpperCase(),
    entity_name: ((formData.get('entity_name') as string) || '').toUpperCase(),
    entity_type: (formData.get('entity_type') as string) || null,
    head_office: (formData.get('head_office') as string) || null,
    website: (formData.get('website') as string) || null,
    industry: (formData.get('industry') as string) || null,
    other_notes: (formData.get('other_notes') as string) || null,
    last_updated: new Date().toISOString(),
  }

  // Client_Source terkunci 'Inbound' kalau Is_From_Lead -- tidak boleh diubah lewat form ini.
  if (!existing.is_from_lead) {
    patch.client_source = formData.get('client_source') as string
  }

  const { error } = await admin.from('client').update(patch).eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath(`/clients/${id}`)
  revalidatePath('/clients')
}

export async function addPic(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const clientId = formData.get('client_id') as string
  const picName = ((formData.get('pic_name') as string) || '').trim()
  if (!picName) throw new Error('Nama PIC wajib diisi.')

  const { data: existingPics } = await admin.from('pic_client').select('id').eq('client_id', clientId)
  const isFirst = (existingPics ?? []).length === 0

  const { error } = await admin.from('pic_client').insert({
    client_id: clientId,
    pic_name: picName,
    title: (formData.get('title') as string) || null,
    email: (formData.get('email') as string) || null,
    phone: (formData.get('phone') as string) || null,
    is_primary: isFirst,
  })
  if (error) throw new Error(error.message)

  revalidatePath(`/clients/${clientId}`)
  revalidatePath('/clients')
}

export async function removePic(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const id = formData.get('id') as string
  const clientId = formData.get('client_id') as string

  const { error } = await admin.from('pic_client').delete().eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath(`/clients/${clientId}`)
  revalidatePath('/clients')
}

export async function setPrimaryPic(formData: FormData) {
  await requireWriteAccess()
  const admin = createAdminClient()
  const id = formData.get('id') as string
  const clientId = formData.get('client_id') as string

  await admin.from('pic_client').update({ is_primary: false }).eq('client_id', clientId)
  const { error } = await admin.from('pic_client').update({ is_primary: true }).eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath(`/clients/${clientId}`)
  revalidatePath('/clients')
}
