import { createClient as createServerClient } from '@/lib/supabase/server'
import { createAdminClient } from '@/lib/supabase/admin'

export async function getCurrentEmployee() {
  const supabase = await createServerClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) return null

  const admin = createAdminClient()

  // Already linked to this login before -- fast path.
  const { data: byAuthId } = await admin.from('employee').select('*').eq('auth_user_id', user.id).maybeSingle()
  if (byAuthId) return byAuthId

  // Not linked yet (e.g. this employee was added via the UI and is logging
  // in for the very first time) -- match by email and self-heal the link
  // so every subsequent login uses the fast path above.
  if (user.email) {
    const { data: byEmail } = await admin.from('employee').select('*').eq('email', user.email).maybeSingle()
    if (byEmail) {
      await admin.from('employee').update({ auth_user_id: user.id }).eq('id', byEmail.id)
      return { ...byEmail, auth_user_id: user.id }
    }
  }

  return null
}

'use server'

import { revalidatePath } from 'next/cache'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'

export async function addMasterDataOption(formData: FormData) {
  const employee = await getCurrentEmployee()
  if (!employee || employee.role !== 'Master Admin') {
    throw new Error('Cuma Master Admin yang boleh menambah Master Data.')
  }

  const category = formData.get('category') as string
  const value = ((formData.get('value') as string) || '').trim()
  const validCategories = ['Head_Office', 'Industry', 'Entity_Type', 'Client_Source']
  if (!validCategories.includes(category)) throw new Error('Category tidak valid.')
  if (!value) throw new Error('Value wajib diisi.')

  const admin = createAdminClient()
  const { error } = await admin.from('master_data').insert({ category, value })
  if (error) throw new Error(error.message)

  revalidatePath('/master-data')
}

'use server'

import { revalidatePath } from 'next/cache'
import { getCurrentEmployee } from '@/lib/get-current-employee'
import { createAdminClient } from '@/lib/supabase/admin'

const ROLES = ['Master Admin', 'Consultant', 'Operation', 'Head of B2B']

async function requireMasterAdmin() {
  const employee = await getCurrentEmployee()
  if (!employee) throw new Error('Belum login.')
  if (employee.role !== 'Master Admin') throw new Error('Cuma Master Admin yang bisa mengelola Employee.')
  return employee
}

async function countOtherActiveMasterAdmins(admin: ReturnType<typeof createAdminClient>, excludeId: string) {
  const { data } = await admin.from('employee').select('id').eq('role', 'Master Admin').eq('status', 'Active')
  return (data ?? []).filter((e: any) => e.id !== excludeId).length
}

export async function addEmployee(formData: FormData) {
  await requireMasterAdmin()
  const admin = createAdminClient()

  const name = ((formData.get('name') as string) || '').trim()
  const email = ((formData.get('email') as string) || '').trim().toLowerCase()
  const role = formData.get('role') as string

  if (!name) throw new Error('Nama wajib diisi.')
  if (!email.endsWith('@kitabisa.com')) throw new Error('Email harus domain @kitabisa.com.')
  if (!ROLES.includes(role)) throw new Error('Role tidak valid.')

  const { error } = await admin.from('employee').insert({ name, email, role, status: 'Active' })
  if (error) {
    if (error.message.includes('duplicate') || error.message.includes('unique')) {
      throw new Error('Email ini sudah terdaftar.')
    }
    throw new Error(error.message)
  }

  revalidatePath('/employees')
}

export async function setEmployeeRole(formData: FormData) {
  await requireMasterAdmin()
  const admin = createAdminClient()
  const id = formData.get('id') as string
  const newRole = formData.get('role') as string
  if (!ROLES.includes(newRole)) throw new Error('Role tidak valid.')

  const { data: emp } = await admin.from('employee').select('*').eq('id', id).single()
  if (!emp) throw new Error('Employee tidak ditemukan.')

  if (emp.role === 'Master Admin' && emp.status === 'Active' && newRole !== 'Master Admin') {
    const others = await countOtherActiveMasterAdmins(admin, id)
    if (others === 0) throw new Error('Tidak bisa mengubah role -- ini satu-satunya Master Admin aktif yang tersisa.')
  }

  const { error } = await admin.from('employee').update({ role: newRole }).eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath('/employees')
}

export async function setEmployeeStatus(formData: FormData) {
  await requireMasterAdmin()
  const admin = createAdminClient()
  const id = formData.get('id') as string
  const newStatus = formData.get('status') as string
  if (!['Active', 'Inactive'].includes(newStatus)) throw new Error('Status tidak valid.')

  const { data: emp } = await admin.from('employee').select('*').eq('id', id).single()
  if (!emp) throw new Error('Employee tidak ditemukan.')

  if (emp.role === 'Master Admin' && emp.status === 'Active' && newStatus === 'Inactive') {
    const others = await countOtherActiveMasterAdmins(admin, id)
    if (others === 0) throw new Error('Tidak bisa menonaktifkan -- ini satu-satunya Master Admin aktif yang tersisa.')
  }

  const { error } = await admin.from('employee').update({ status: newStatus }).eq('id', id)
  if (error) throw new Error(error.message)

  revalidatePath('/employees')
}
