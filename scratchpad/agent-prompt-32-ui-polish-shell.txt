'use client'

import Link from 'next/link'
import { usePathname } from 'next/navigation'

const NAV_GROUPS = [
  {
    label: 'Utama',
    items: [
      { href: '/dashboard', label: 'Dashboard Sales' },
      { href: '/', label: 'Home' },
    ],
  },
  {
    label: 'Sales',
    items: [
      { href: '/leads', label: 'Lead Capturing' },
      { href: '/clients', label: 'Client Monitoring' },
      { href: '/projects', label: 'Sales Pipeline' },
    ],
  },
  {
    label: 'Operasional',
    items: [
      { href: '/documents', label: 'Document Pipeline' },
      { href: '/cor-entities', label: 'Vendor' },
      { href: '/cost-monitoring', label: 'Cost Monitoring' },
    ],
  },
  {
    label: 'Pengaturan',
    items: [
      { href: '/achievement-target', label: 'Achievement Target' },
      { href: '/employees', label: 'Employee' },
      { href: '/master-data', label: 'Master Data' },
    ],
  },
]

export function SidebarNav() {
  const pathname = usePathname()

  function isActive(href: string) {
    if (href === '/') return pathname === '/'
    return pathname === href || pathname.startsWith(href + '/')
  }

  return (
    <nav className="flex flex-col gap-6">
      {NAV_GROUPS.map((group) => (
        <div key={group.label}>
          <div className="mb-2 px-3 text-[11px] font-semibold uppercase tracking-wider text-slate-400">
            {group.label}
          </div>
          <div className="flex flex-col gap-0.5">
            {group.items.map((item) => {
              const active = isActive(item.href)
              return (
                <Link
                  key={item.href}
                  href={item.href}
                  className={`rounded-md px-3 py-2 text-sm font-medium transition-colors ${
                    active
                      ? 'bg-indigo-50 text-indigo-700'
                      : 'text-slate-600 hover:bg-slate-100 hover:text-slate-900'
                  }`}
                >
                  {item.label}
                </Link>
              )
            })}
          </div>
        </div>
      ))}
    </nav>
  )
}
